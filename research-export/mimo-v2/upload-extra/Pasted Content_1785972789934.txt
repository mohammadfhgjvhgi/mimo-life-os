        <defs>

            <stop offset="100%" stopColor="#7c3aed" stopOpacity="0" />
          </linearGradient>
        </defs>
        {[0.25, 0.5, 0.75].map((f) => (
          <line key={f} x1="0" x2={W} y1={H * f} y2={H * f} stroke="var(--bd)" strokeDasharray="4 4" strokeWidth="1" />

        <polygon points={area} fill="url(#ag)" />

const DonutSVG = ({ data }) => {

  const C = 2 * Math.PI * 40;
  let acc = 0;
  return (
    <div style={{ display: 'flex', alignItems: 'center', gap: 18 }}>
      <svg viewBox="0 0 100 100" style={{ width: 150, height: 150, flexShrink: 0 }}>
        {data.map((x, i) => {
          const frac = x.value / total;
          const el = (
            <circle key={i} cx="50" cy="50" r="40" fill="none" stroke={x.c} strokeWidth="14"
              strokeDasharray={`${Math.max(frac * C - 2, 0)} ${C}`} strokeDashoffset={-acc * C}
              transform="rotate(-90 50 50)" />

const RadarSVG = ({ data }) => {
  const cx = 60, cy = 55, R = 38;
  const pt = (i, r) => {
    const a = (Math.PI * 2 * i) / data.length - Math.PI / 2;
    return [cx + Math.cos(a) * r, cy + Math.sin(a) * r];
  };
  const valPts = data.map((d, i) => pt(i, (R * d.v) / 100).join(',')).join(' ');
  return (
    <svg viewBox="0 0 120 110" style={{ width: '100%', maxWidth: 260, margin: '0 auto', display: 'block' }}>
      {[0.33, 0.66, 1].map((f) => (
        <polygon key={f} points={data.map((_, i) => pt(i, R * f).join(',')).join(' ')} fill="none" stroke="var(--bd)" strokeWidth="1" />
      ))}
      {data.map((_, i) => {
        const [x, y] = pt(i, R);
        return <line key={i} x1={cx} y1={cy} x2={x} y2={y} stroke="var(--bd)" strokeWidth="1" />;
      })}
      <polygon points={valPts} fill="#a855f7" fillOpacity=".3" stroke="#a855f7" strokeWidth="2" />

      })}

                placeholder={mode === 'research' ? 'ما الذي تريد البحث عنه بعمق؟' : mode === 'code' ? 'صف ما تريد بناءه...' : 'اسأل Nova Ultra أي شيء...'} />
              <div className="tools">
                <button className="tbtn" title="إرفاق ملف" onClick={() => toast('رفع الملفات قريباً 📎')}><I.Paperclip size={17} /></button>
                <button className="tbtn" title="توليد صورة" onClick={() => setImgGen(true)}><I.Image size={17} /></button>
                <button className={'tbtn' + (deepThink ? ' on' : '')} title="تفكير عميق" onClick={() => setDeepThink(!deepThink)}><I.Brain size={17} /></button>
                <button className={'tbtn' + (webSearch ? ' on' : '')} title="بحث الويب" onClick={() => setWebSearch(!webSearch)}><I.Globe size={17} /></button>
                <button className="tbtn" title="وضع الصوت" onClick={() => setVoice(true)}><I.Mic size={17} /></button>
                <button className="tbtn" title="لوحة الأوامر" onClick={() => setPalette(true)}><I.Command size={17} /></button>
                {loading ? <button className="sendb stop" onClick={stop}><I.Stop size={16} /></button>
                : <button className="sendb" onClick={() => send()} disabled={!input.trim()}><I.Send size={16} style={{ transform: 'scaleX(-1)' }} /></button>}
              </div>
            </div>
            <div className="hint"><b>{curModel.name}</b> قد يخطئ أحياناً · Ctrl+K للأوامر · Enter للإرسال</div>
          </div>
        )}
      </main>

      {/* ═══ Artifacts Panel ═══ */}
      <AnimatePresence>
        {panelOpen && artifact && (
          <motion.aside className="rp" initial={{ marginLeft: -430, opacity: 0 }} animate={{ marginLeft: 0, opacity: 1 }} exit={{ marginLeft: -430, opacity: 0 }} transition={{ duration: .25 }}>
            <div className="rp-h">
              <span style={{ width: 28, height: 28, borderRadius: 8, background: 'var(--gsoft)', color: 'var(--pr)', display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
                {artifact.type === 'code' ? <I.Code size={15} /> : artifact.type === 'run' ? <I.Terminal size={15} /> : <I.Canvas size={15} />}
              </span>
              <span className="tt">{artifact.title}</span>
              <button className="icb" style={{ padding: 6 }} onClick={() => { if (navigator.clipboard) navigator.clipboard.writeText(artifact.code || artifact.content); toast('تم النسخ ✓'); }}><I.Copy size={15} /></button>
              <button className="icb" style={{ padding: 6 }} onClick={() => setPanelOpen(false)}><I.X size={15} /></button>
            </div>
            <div className="rp-b">
              {artifact.type === 'code' && (
                <div>
                  <div className="codeblk">
                    <div className="code-h"><span style={{ fontWeight: 800 }}>{artifact.lang}</span><span>{artifact.code.split('\n').length} سطر</span></div>
                    <div className="code-b">{artifact.code}</div>
                  </div>
                  <div style={{ display: 'flex', gap: 8 }}>
                    <button className="newchat" style={{ flex: 1 }} onClick={() => toast('تشغيل الكود... ⚡')}><I.Play size={14} /> تشغيل</button>
                    <button className="pill" onClick={() => toast('تم الحفظ ✓')}><I.Download size={14} /></button>
                  </div>
                </div>
              )}
              {artifact.type === 'run' && (
                <div>
                  <div className="codeblk">
                    <div className="code-h"><span style={{ fontWeight: 800 }}>الكود</span></div>
                    <div className="code-b">{artifact.code}</div>
                  </div>
                  <div className="termblk">
                    <div style={{ color: '#8b949e', marginBottom: 8 }}>➜ node script.js</div>
                    {artifact.output.split('\n').map((l, i) => <div key={i}>{l}</div>)}
                  </div>
                </div>
              )}
              {artifact.type === 'doc' && (
                <div style={{ background: 'var(--bg3)', border: '1px solid var(--bd)', borderRadius: 13, padding: 16 }}>
                  <div className="aict">{renderContent(artifact.content)}</div>
                </div>
              )}
            </div>
          </motion.aside>
        )}
      </AnimatePresence>

      {/* ═══ Command Palette ═══ */}
      <AnimatePresence>{palette && (
        <motion.div className="pal" initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} onClick={() => setPalette(false)}>
          <motion.div className="pal-card" initial={{ scale: .96, y: -10 }} animate={{ scale: 1, y: 0 }} exit={{ scale: .96, y: -10 }} onClick={(e) => e.stopPropagation()}>
            <div className="pal-in"><I.Command size={16} style={{ color: 'var(--pr)' }} /><input autoFocus value={palQ} onChange={(e) => setPalQ(e.target.value)} placeholder="اكتب أمراً... (محادثة، تحليلات، مهام...)" /><kbd style={{ fontSize: 10, color: 'var(--tx3)', background: 'var(--bg3)', padding: '2px 6px', borderRadius: 5 }}>ESC</kbd></div>
            <div style={{ maxHeight: 340, overflowY: 'auto' }}>
              {PAL.map((a, i) => {
                const PalIcon = a.ic;
                return <div key={i} className="pal-item" onClick={() => { a.fn(); setPalette(false); setPalQ(''); }}><PalIcon size={16} />{a.l}{a.k && <kbd>{a.k}</kbd>}</div>;
              })}
            </div>
          </motion.div>
        </motion.div>
      )}</AnimatePresence>

      {/* ═══ Voice Mode ═══ */}
      <AnimatePresence>{voice && (
        <motion.div className="voice" initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}>
          <motion.div className="orb" animate={{ scale: [1, 1.08, 1] }} transition={{ duration: 2, repeat: Infinity }}><img src={LOGO} alt="" /></motion.div>
          <div className="wave">{Array.from({ length: 14 }).map((_, i) => <span key={i} style={{ animationDelay: (i * .08) + 's', height: 20 + (i % 5) * 9 }} />)}</div>
          <div className="st">Nova يستمع ويتحدث...</div>
          <motion.div className="tx" key={voiceLine} initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }}>{voiceLine}</motion.div>
          <button className="sendb stop" style={{ marginTop: 30, width: 52, height: 52, borderRadius: '50%' }} onClick={() => setVoice(false)}><I.Stop size={20} /></button>
        </motion.div>
      )}</AnimatePresence>

      {/* ═══ Image Gen ═══ */}
      <AnimatePresence>{imgGen && (
        <motion.div className="modal" initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} onClick={() => setImgGen(false)}>
          <motion.div className="modal-card" initial={{ scale: .95, y: 14 }} animate={{ scale: 1, y: 0 }} exit={{ scale: .95, y: 14 }} onClick={(e) => e.stopPropagation()}>
            <div className="modal-h"><I.Image size={18} style={{ color: 'var(--pr)' }} /><span className="tt">توليد الصور بالذكاء الاصطناعي</span><button className="icb" onClick={() => setImgGen(false)}><I.X size={16} /></button></div>
            <div className="modal-b">
              <button className="newchat" style={{ marginBottom: 16 }} onClick={genImage} disabled={genning}>{genning ? 'جارٍ التوليد...' : '✨ توليد صورة جديدة'}</button>
              <div className="imggrid">
                {genImgs.map((src, i) => (
                  <div key={i} className="imgcard">{genning && i === 0 ? <div className="gen"><motion.span animate={{ rotate: 360 }} transition={{ duration: 1, repeat: Infinity, ease: 'linear' }} style={{ display: 'flex' }}><I.Refresh size={22} /></motion.span></div> : <img src={src} alt="" />}</div>
                ))}
              </div>
            </div>
          </motion.div>
        </motion.div>
      )}</AnimatePresence>

      {/* ═══ Settings ═══ */}
      <AnimatePresence>{settings && (
        <motion.div className="modal" initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} onClick={() => setSettings(false)}>
          <motion.div className="modal-card" initial={{ scale: .95, y: 14 }} animate={{ scale: 1, y: 0 }} exit={{ scale: .95, y: 14 }} onClick={(e) => e.stopPropagation()}>
            <div className="modal-h"><I.Settings size={18} style={{ color: 'var(--pr)' }} /><span className="tt">الإعدادات</span><button className="icb" onClick={() => setSettings(false)}><I.X size={16} /></button></div>
            <div className="modal-b">
              {[{ l: 'التفكير العميق افتراضياً', d: deepThink, f: () => setDeepThink(!deepThink) },
                { l: 'بحث الويب التلقائي', d: webSearch, f: () => setWebSearch(!webSearch) },
                { l: 'الوضع الداكن', d: theme === 'dark', f: () => setTheme(theme === 'dark' ? 'light' : 'dark') }].map((s, i) => (
                <div key={i} style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', padding: '13px 0', borderBottom: '1px solid var(--bd)' }}>
                  <span style={{ fontSize: 13.5, fontWeight: 600 }}>{s.l}</span>
                  <button onClick={s.f} style={{ width: 46, height: 26, borderRadius: 20, border: 'none', cursor: 'pointer', background: s.d ? 'var(--grad)' : 'var(--bg4)', position: 'relative', transition: 'background .2s' }}>
                    <motion.span layout transition={{ type: 'spring', stiffness: 500, damping: 30 }} style={{ position: 'absolute', top: 3, width: 20, height: 20, borderRadius: '50%', background: '#fff', right: s.d ? 3 : 23 }} />
                  </button>
                </div>
              ))}
              <div style={{ marginTop: 16, fontSize: 12, color: 'var(--tx3)', textAlign: 'center' }}>Nova Ultra v4.0 · صُنع بواسطة محمد عادل 🇵🇸</div>
            </div>
          </motion.div>
        </motion.div>
      )}</AnimatePresence>

      {/* ═══ Toasts ═══ */}
      <div className="toasts">
        <AnimatePresence>{toasts.map((t) => (
          <motion.div key={t.id} className="toast" initial={{ opacity: 0, y: 14, scale: .95 }} animate={{ opacity: 1, y: 0, scale: 1 }} exit={{ opacity: 0, scale: .95 }}><I.Sparkles size={14} style={{ color: 'var(--pr)' }} />{t.t}</motion.div>
        ))}</AnimatePresence>
      </div>
    </div>
  );
}
